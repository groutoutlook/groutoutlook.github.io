           /**
* Lets page scripts ask the connector to save the current page, e.g. a Vimium C key mapping that
* runs `document.dispatchEvent(new Event('zotero-connector-save'))`. This takes the same path as
* clicking the toolbar button.
*/
document.addEventListener('zotero-connector-save', function () {
 	Zotero.Connector_Browser.savePage();
});
  // Vimium C's sendToExtension avoids javascript: URLs, which may be blocked by the page.
  this.onMessageExternal = function(message, sender, sendResponse) {
  	if (sender.id !== 'hfjbmagddngcpeloejdejnfgbamkjaeg' || message !== 'savePage') {
  		return false;
  	}
  	(async () => {
  		await Zotero.initDeferred.promise;
  		await Zotero.Connector_Browser.savePage(null, await getCurrentTab());
  		sendResponse(true);
  	})().catch(error => {
  		Zotero.logError(error);
  		sendResponse('Zotero could not save this page. Try the Zotero toolbar button.');
  	});
  	return true;
  }
