    • Run: #AI taught me this. Convert int -> hex -> transform each octet into a IP address. 
      $timeout = 300
$arpTable = @()
foreach ( $index in 1..255){
    $dns = [System.Net.Dns]::GetHostEntryAsync("192.168.110.$index")
    $dns.Wait($timeout) | Out-Null
    if ($dns.IsCompleted) {
        if ($dns.Status -ne "Faulted"){
            $arpTable += $dns.Result
        }
    } else { Write-Error "DNS query timed out" }
}
$arpTable

(tohex $arpTable[0].AddressList[0].Address) -split '(.{2})' -ne ''| % {[System.Convert]::ToInt32($_,16)} -ob 3| tee -Variable resolvedIP
$resolvedIP -join "."
