pkg i rsync
cd ~/..
mkdir usr-n && cd usr-n
curl -L https://github.com/termux-pacman/termux-packages/releases/download/bootstrap-2026.04.26-r1%2Bpacman.android-7/bootstrap-aarch64.zip -o ./bootstrap
unzip ./bootstrap
cat SYMLINKS.txt | awk -F "←" '{system("ln -s '"'"'"$1"'"'"' '"'"'"$2"'"'"'")}'
cd ~/..
# INFO: at this point launch another shell that bundles all of the utils as `nu`
# NOTE: or you could open `failsafe` to run `/system/bin/sh`.
# rm -fr usr/
# mv usr-n/ usr/
rsync -avh --progress ./usr-n ./usr
pacman-key --init
pacman-key --populate
sh -c "$(curl -L https://groutoutlook.github.io/bootstrap_termux.sh)"
chsh -s nu
