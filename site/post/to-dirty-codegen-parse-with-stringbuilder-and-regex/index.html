      function Show-PortGapsWithRenderedLUT {
    param(
        [string]$Raw = (Get-Clipboard)
    )
    $pattern = '(?s)(Port\d+LUT_RGB)\[\]\s*=\s*\{(.*?)\};'
    $matches = [regex]::Matches($Raw, $pattern)
    if ($matches.Count -eq 0) { return }

    $blocks = foreach ($match in $matches) {
        $label   = $match.Groups[1].Value
        $values  = $match.Groups[2].Value -split '[^\d]+' | Where-Object { $_ } | ForEach-Object { [int]$_ }


        $gaps = New-Object System.Collections.Generic.List[int]
        for ($i = 0; $i -lt $values.Count - 1; $i++) {
            if ([math]::Abs($values[$i] - $values[$i + 1]) -gt 1) {
                $gaps.Add($values[$i + 1])
                $i++
            }
        }

        [pscustomobject]@{
            Label = $label
            Gaps  = $gaps
        }
    }

    $sb = [System.Text.StringBuilder]::new()
    foreach ($block in $blocks) {
        [void]$sb.AppendLine("            # $($block.Label)")
        [void]$sb.AppendLine('            {')
        foreach ($n in $block.Gaps) {
            [void]$sb.AppendLine("                $n :0,")
        }
        [void]$sb.AppendLine('            },')
        [void]$sb.AppendLine()
    }

    $sb.ToString() | Set-Clipboard
    $sb.ToString()
}
