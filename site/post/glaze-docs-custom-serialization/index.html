    • Recap: custom read/write for specific fields by two lambdas.
    • Run: 
      struct custom_encoding{
  uint64_t x,y,z;
  std::string str;
  void read_x(const std::string& s) {
     x = std::stoi(s);
  }
  
  uint64_t write_x() {
     return x;
  }
}
template <>
struct glz::meta<custom_encoding>
{
   using T = custom_encoding;
   static constexpr auto read_s = [](T& s, const std::string& input) { s.str = input; };
   static constexpr auto write_s = [](auto& s) -> auto& { return s.str; };
   static constexpr auto value = object("x", custom<&T::read_x, &T::write_x>, //
                                        "y", custom<&T::read_y, &T::y>, //
                                        "z", custom<&T::z, &T::write_z>
                                        "s", custom<&T::s,write_s>);
};
