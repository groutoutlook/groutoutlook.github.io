      $SquareBracketsParameter = @{
    # PSReadLine identifies Ctrl+[ as Ctrl+Oem4 on Windows consoles.
    Key              = 'Ctrl+Oem4'
    BriefDescription = 'brackets the selection or nearest token'
    LongDescription  = '`ctrl+[` Wraps selected text in square brackets; if no selection, wraps the token nearest to the cursor. Cursor is placed after the closing bracket.'
    ScriptBlock      = {
        param($key, $arg)
        Invoke-PSReadLineWrapToken -Open '[' -Close ']'
    }
}

$BacktickParameter = @{
    # PSReadLine read this as Ctrl+shift+`
    Key              = 'Ctrl+Shift+Oem3'
    BriefDescription = 'backtick the selection or nearest token'
    LongDescription  = '`Ctrl+shift+backtick` Wraps selected text in backticks; if no selection, wraps the token nearest to the cursor. Cursor is placed after the closing backtick.'
    ScriptBlock      = {
        param($key, $arg)
        Invoke-PSReadLineWrapToken -Open '`' -Close '`'
    }
}
