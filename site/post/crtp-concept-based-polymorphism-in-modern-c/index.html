    • Recap:
    • Run: compile time/ static polymorphism. Good dynamic with deducing this which is C++23 up. 
      struct Base { void name(this auto&& self) { self.impl(); } };
struct D1 : public Base { void impl() { std::puts("D1::impl()"); } };
struct D2 : public Base { void impl() { std::puts("D2::impl()"); } };
// now D1 can have its name() method ahead of time due to compiler deducing the name of it.
template<typename Derived>
void print(Derived const& instance) {
    instance.name();
}
main{
  // quite magical.
  D1 d1; print(d1);
  D2 d2; print(d2);
}
