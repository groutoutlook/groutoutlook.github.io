import FreeCAD as App

def coord_key(obj, places=6):
    if hasattr(obj, "Point"):
        p = obj.Point
        return tuple(round(v, places) for v in (p.x, p.y, p.z))
    if hasattr(obj, "Placement"):
        base = obj.Placement.Base
        return tuple(round(v, places) for v in (base.x, base.y, base.z))
    if hasattr(obj, "Shape") and obj.Shape.Vertexes:
        v = obj.Shape.Vertexes[0].Point
        return tuple(round(vv, places) for vv in (v.x, v.y, v.z))
    return None

doc = App.ActiveDocument
seen = {}
for obj in list(doc.Objects):
    key = coord_key(obj)
    if key is None:
        continue
    if key in seen:
        doc.removeObject(obj.Name)
    else:
        seen[key] = obj

doc.recompute()
