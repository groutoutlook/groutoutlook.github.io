      javascript:(function(){
    var links = document.querySelectorAll('a');
    var count = 0;
    links.forEach(function(l){
        /* Skip javascript links or empty anchors */
        if(l.href && !l.href.startsWith('javascript:') && l.getAttribute('href') !== '#'){
            l.setAttribute('target', '_blank');
            l.setAttribute('rel', 'noopener');
            count++;
        }
    });
    /* Show a small temporary message in top right corner */
    var msg = document.createElement('div');
    msg.innerText = count + " Links set to New Tab";
    msg.style.cssText = 'position:fixed;top:10px;right:10px;background:#d00;color:#fff;padding:8px 12px;z-index:2147483647;border-radius:4px;font-family:sans-serif;font-weight:bold;box-shadow:0 2px 5px rgba(0,0,0,0.3);';
    document.body.appendChild(msg);
    setTimeout(function(){ msg.remove(); }, 2000);
})();
