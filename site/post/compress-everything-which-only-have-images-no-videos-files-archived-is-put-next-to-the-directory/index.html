      # Run from the root where you want to search (or set $root)
$root = Get-Location

Get-ChildItem -Path $root -Recurse -File -Include *.jpg,*.jpeg,*.png,*.webp,*.jxl,*.gif,*.mp4,*.webm |
  Group-Object DirectoryName |
  Where-Object {
    $exts = ($_.Group.Extension | ForEach-Object { $_.ToLowerInvariant() })
    # has JPG|JPEG|PNG|WEBP|JXL, and does NOT contain gif/mp4/webm
    (($exts -contains '.jpg' -or $exts -contains '.jpeg' -or $exts -contains '.png' -or $exts -contains '.webp' -or $exts -contains '.jxl') -and
    -not ($exts -contains '.gif' -or $exts -contains '.mp4' -or $exts -contains '.webm'))
  } |
  ForEach-Object {
    $dirPath = $_.Name
    $zipName = "{0}.zip" -f (Split-Path $dirPath -Leaf)
    $zipPath = Join-Path -Path (Split-Path $dirPath -Parent) -ChildPath $zipName

    if (Test-Path -Path $zipPath -PathType Leaf) {
        Write-Output "ZIP already exists - skipping: $zipPath"
        return  # skip this directory
    }

    Compress-Archive -Path (Join-Path $dirPath '*') -DestinationPath $zipPath -Force
    Write-Output "Created: $zipPath"
  }
