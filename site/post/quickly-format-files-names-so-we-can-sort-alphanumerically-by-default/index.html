      Get-ChildItem -File -Recurse |
  Where-Object { $_.Extension -match '^\.(png|jpe?g)$' -and $_.BaseName -match '^\d+$' } |
  ForEach-Object {
    $newName = ('{0:D4}' -f [int]$_.BaseName) + $_.Extension.ToLower()
    Rename-Item -LiteralPath $_.FullName -NewName $newName
  }
