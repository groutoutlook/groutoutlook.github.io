int shortest_distance(vector<int> &A) {
    if(A.size() == 2){
        return abs(A[0] - A[1]);
    }
    auto a = 1;
    int left_val = std::accumulate(A.begin(), A.begin()+a, 0);
    int right_val = std::accumulate(A.begin()+a, A.end(), 0);
    auto ans = min(ans, abs(left_val-right_val));
    for(auto a = 1; a < A.size() - 1; a++){
        left_val += A[a];
        right_val -= A[a];
