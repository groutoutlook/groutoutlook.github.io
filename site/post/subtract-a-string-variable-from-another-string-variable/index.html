      # get git root then subtract the path so we can get the subproject name.
$current_gitroot = Split-Path (git rev-parse --show-toplevel)
$current_proj = $pwd.path.replace("$current_gitroot\","").replace('\','.')
Write-Host "Current project : $current_proj" -Fore Yellow
