    • Recap: you cant specify input like, find match in string as -match either,.

      • Return types: MatchInfo
    • Run: you cant directly replace strings with sls either 
      "a string" | Select-String -Pattern "str" | % { $_.Line.Replace($_.Matches.Value, "g") }

# extracting hexadecimal like string in clipboard.
gcb | sls -all "[0-9a-fA-F]{3,}" |%{ $_.Matches.Value}

# or just...playing music from jrnl...
jrnl ms -c lol | sls "http.*\)" |%{mpv $_.Matches.Value}
# It's less annoying changing position in the GUI everytime tracks changed.
rgj le sserafim 3 |sls "http.*\)"|%{$_.Matches.Value -replace '\)'}|Split-Batch 10|%{mpv $_}    
while(ps -name mpv){Sleep -Sec 5;Write-Host "INFO: [$(Get-Date)] mpv running"} ; cd rare audio && rr
