    function Write-UniqueHistory(){
  $historyPath = (Get-PSReadlineOption).HistorySavePath
  $history = Get-Content -Path $historyPath
  $uniqueHistory = [System.Collections.Generic.List[string]]::new()
  $seen = [System.Collections.Generic.HashSet[string]]::new()
  # Process lines in reverse to keep the most recent occurrence
  for ($i = $history.Count - 1; $i -ge 0; $i--) {
      $line = $history[$i]
      if ($seen.Add($line)) {
          $uniqueHistory.Add($line)
      }
  }
  # Reverse back to maintain original order (most recent at the bottom)
  $uniqueHistory.Reverse()
  # Save the deduplicated history back to the file
  $uniqueHistory | Set-Content -Path $historyPath -Force
  Write-Output "Deduplicated history file. Original lines: $($history.Count), Unique lines: $($uniqueHistory.Count)"
}
