auto find_odd_out(vector<int> &A) {
std::unordered_map<int,int> recorded{};
    int temp_ans = 0;
    for (auto &i : A)
    {
        auto temp_val = recorded[i];
        recorded[i] = temp_val + 1;
    }

    for (auto & [val, count] : recorded){
        if (count%2 == 1) return val;
    }
}

int solution(vector<int> &A) {

int miss_finding(vector<int> &A)
{
    unordered_map<int,int> record;
    for (auto &i : A){
        record[i] = i;
    }
    if (A.size() == 0){
        return 1;
    }
    for (auto i = 0; i <= A.size(); i+=1){
        if(record[i+1] == 0) return (i+1);
    }
}
int solution(vector<int> &A) {
    auto a = miss_finding(A);
    return a;
