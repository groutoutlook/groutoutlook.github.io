import FreeCAD, FreeCADGui

def select_at(x, y, tol=1.0):
    FreeCADGui.Selection.Filter("") # Clear any active filters
    doc = FreeCAD.ActiveDocument
    target = FreeCAD.Vector(x, y, 0)
    
    for o in doc.Objects:
        if not hasattr(o, "Shape"): continue
        p = o.Shape.CenterOfMass
        dist = (target - FreeCAD.Vector(p.x, p.y, 0)).Length
        
        if dist < tol:
            FreeCADGui.Selection.addSelection(o)
            print(f"Selected {o.Name} (dist: {dist:.2f})")
            return
            
    print(f"Nothing found at {x:.2f}, {y:.2f}")
