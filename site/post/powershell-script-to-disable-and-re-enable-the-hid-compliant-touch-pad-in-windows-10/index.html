$mousePadId = Get-PnpDevice | where {$_.FriendlyName -like '*touch pad*'} | %{$_.InstanceId}
Disable-PnpDevice -InstanceId $mousePadId -Confirm:$false

# to re-enable
Enable-PnpDevice -Ins $mousePadId
