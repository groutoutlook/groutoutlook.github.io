def move_selection(dx, dy, doc=None):
    if doc is None:
        doc = App.ActiveDocument
    if doc is None:
        raise ValueError("No active document found.")
    global aa
    selection = Gui.Selection.getSelection()
    aa = selection.copy()
    doc.openTransaction("move selection")
    try:
        for obj in selection:
            if hasattr(obj, "Placement") and obj.Document == doc:
                base = obj.Placement.Base
                doc.getObject(obj.Name).Placement = App.Placement(
                    App.Vector(base.x + dx, base.y + dy, base.z),
                    obj.Placement.Rotation,
                    App.Vector(0, 0, 0),
                )
        doc.recompute()
    finally:
        doc.commitTransaction()

move_selection(0, 30)
