      Start-Job -ScriptBlock {
    Start-Sleep -Seconds 2
    & 'smartcontextmenu' --titleEnds 'mpv' --transparency 20 --alwaysontop on --hidealttab on --clickthrough on --priority realtime
}
mpv --playlist=$playlistDir --loop-playlist=1 --panscan=1.0 --profile=fast --sub-pos=20 --sub-color=1.0/0.0/0.0/0.75  --lavfi-complex='[aid1] asplit [ao] [v] ; [v] showwaves=mode=line:split_channels=1 [vo]' --force-window
