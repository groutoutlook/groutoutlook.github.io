      cifm-parent = "!f() { root=$(git rev-parse --show-toplevel) || return 1; cd \"$root\" || return 1; path=$1; shift; message=$*; [ -n \"$path\" ] && [ -n \"$message\" ] || { echo 'usage: git cifm-parent <path> <message>'; return 2; }; blob=$(git hash-object -w -- \"$path\") || return 1; git update-index --add --cacheinfo \"100644,$blob,$path\" || return 1; git commit --only -m \"$message\" -- \"$path\" && git puf; }; f"
